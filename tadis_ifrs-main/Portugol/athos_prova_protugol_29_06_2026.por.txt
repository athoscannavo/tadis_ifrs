	funcao inteiro localizar(cadeia nomesp[], inteiro quant[], cadeia nomep, inteiro limite)
	{
		inteiro i = 0
		para(i=0;i<limite;i++)
		{
			se(nomesp[i] == nomep)
			{
				retorne i
			}
				
		}

	}

	}
